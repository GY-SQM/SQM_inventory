  COUNT(*) as tonbag_count,
  SUM(qty_kg) as total_kg,
  MIN(picking_date) as picking_date
FROM picking_table
WHERE status='ACTIVE'
GROUP BY lot_no, picking_no
ORDER BY picking_date DESC
